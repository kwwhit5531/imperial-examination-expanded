# ROOT/this: candidate. scope:activity comes from the vanilla activity callback.
# This is the single tier/phase gate to broaden after the local prototype passes.
iee_is_exam_candidate_trigger = {
    has_tgp_dlc_trigger = yes
    is_alive = yes
    exists = scope:activity
    involved_activity ?= scope:activity
    location = scope:activity.activity_location
    is_in_guest_subset = { name = entrants }
    has_activity_intent = local_exam_taker_intent
    scope:activity = {
        has_activity_type = activity_local_examination
        has_current_phase = local_examination_phase_examination
        is_current_phase_active = yes
    }
}

iee_narrative_available_trigger = {
    iee_is_exam_candidate_trigger = yes
    NOT = { has_character_flag = iee_narrative_spacing }
    NOT = { has_variable = been_caught_cheating }
}

# Current character is the prospective peer/examiner; ROOT remains the recipient.
# Guest lists may contain characters who have already left the activity.
iee_valid_other_attendee_trigger = {
    this != root
    is_alive = yes
    involved_activity ?= scope:activity
    location = scope:activity.activity_location
}

iee_valid_peer_trigger = {
    iee_valid_other_attendee_trigger = yes
    is_in_guest_subset = { name = entrants }
    NOT = { has_variable = been_caught_cheating }
}

iee_valid_examiner_trigger = {
    iee_valid_other_attendee_trigger = yes
    is_in_guest_subset = { name = imperial_examiners }
}

iee_has_peer_trigger = {
    scope:activity = {
        any_guest_subset = {
            name = entrants
            iee_valid_peer_trigger = yes
        }
    }
}

iee_has_examiner_trigger = {
    scope:activity = {
        any_guest_subset = {
            name = imperial_examiners
            iee_valid_examiner_trigger = yes
        }
    }
}

# ASSISTANT ELIGIBILITY: this checklist permits events, but does not cause one by itself.
iee_assistant_available_trigger = {
    # Require the All Under Heaven DLC used by the examination system.
    has_tgp_dlc_trigger = yes
    # The recipient must still be alive.
    is_alive = yes
    # An examination activity must have been supplied by the callback.
    exists = scope:activity
    # The recipient must still belong to this exact activity; ?= handles a missing reference safely.
    involved_activity ?= scope:activity
    # Require physical presence, excluding travel and characters who have left.
    location = scope:activity.activity_location
    # Require the actual Assist Examination intent, not a prior passing result.
    has_activity_intent = exam_assist_exam_intent
    # Exclude registered candidates, even if their intent has changed.
    NOT = { is_in_guest_subset = { name = entrants } }
    # Wait until the 14-day break after any previous mod encounter has expired.
    NOT = { has_character_flag = iee_narrative_spacing }
    # The following checks inspect the activity rather than the character.
    scope:activity = {
        # Limit this expansion to local examinations.
        has_activity_type = activity_local_examination
        # Exclude phase transitions; the current phase must actually be active.
        is_current_phase_active = yes
        # Either of these two phases is allowed. The examination itself is excluded.
        OR = {
            # Assembly: arrivals and initial arrangements.
            has_current_phase = local_examination_phase_assembly
            # Preparation: getting ready before the actual papers.
            has_current_phase = local_examination_phase_preparation
        }
    }
}
